cmake_minimum_required(VERSION 3.16)
project(VIGOR_paper_benchmarks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type")
endif()
find_package(Threads REQUIRED)
find_package(OpenMP REQUIRED)
find_package(Boost REQUIRED)
set(PAPER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")

add_library(paper_support INTERFACE)
target_include_directories(paper_support INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/run" ${Boost_INCLUDE_DIRS})
target_link_libraries(paper_support INTERFACE Threads::Threads OpenMP::OpenMP_CXX)
target_compile_definitions(paper_support INTERFACE PAPER_ROOT="${PAPER_ROOT}")
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(paper_support INTERFACE -O3 -fno-omit-frame-pointer)
    if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64")
        target_compile_options(paper_support INTERFACE -mno-avx -mno-avx2 -mno-fma -fno-fast-math)
    endif()
endif()

add_executable(benchmark_vigor run/benchmark_vigor.cpp)
target_link_libraries(benchmark_vigor PRIVATE paper_support)
target_include_directories(benchmark_vigor PRIVATE "${PAPER_ROOT}/include")

# Native baselines share an initial-index constructor built from hnswlib.
# VIGOR has no dependency on these sources or the Python header-preparation step.
if(EXISTS "${PAPER_ROOT}/refs/hnswlib/hnswlib/hnswalg.h")
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    function(prepare_vendor engine source)
        cmake_parse_arguments(VENDOR "" "OVERLAY" "" ${ARGN})
        set(output "${CMAKE_CURRENT_BINARY_DIR}/headers/${engine}/hnswlib")
        set(args --source "${source}" --output "${output}")
        if(VENDOR_OVERLAY)
            list(APPEND args --overlay "${VENDOR_OVERLAY}")
            set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${VENDOR_OVERLAY}")
        endif()
        if(engine STREQUAL "wolverine")
            list(APPEND args --wolverine)
        endif()
        set(headers hnswlib.h hnswalg.h space_l2.h space_ip.h visited_list_pool.h bruteforce.h stop_condition.h)
        foreach(header IN LISTS headers)
            set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${source}/${header}")
        endforeach()
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/run/prepare_native.py")
        execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/run/prepare_native.py" ${args}
                        RESULT_VARIABLE result ERROR_VARIABLE error)
        if(NOT result EQUAL 0)
            message(FATAL_ERROR "Cannot prepare ${engine}: ${error}")
        endif()
    endfunction()
    prepare_vendor(hnsw "${PAPER_ROOT}/refs/hnswlib/hnswlib")
    add_library(paper_native_initial STATIC run/native_initial.cpp)
    target_link_libraries(paper_native_initial PRIVATE paper_support)
    target_include_directories(paper_native_initial PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/headers/hnsw")
    set(native_engines hnsw)
    if(EXISTS "${PAPER_ROOT}/refs/PRO-HNSW/src/hnswalg.h")
        prepare_vendor(prohnsw "${PAPER_ROOT}/refs/hnswlib/hnswlib"
                       OVERLAY "${PAPER_ROOT}/refs/PRO-HNSW/src/hnswalg.h")
        list(APPEND native_engines prohnsw)
    else()
        message(STATUS "PRO-HNSW sources absent; skipping benchmark_prohnsw")
    endif()
    if(EXISTS "${PAPER_ROOT}/refs/Wolverine/hnsw_Wolverine/hnswalg.h")
        prepare_vendor(wolverine "${PAPER_ROOT}/refs/Wolverine/hnsw_Wolverine")
        list(APPEND native_engines wolverine)
    else()
        message(STATUS "Wolverine sources absent; skipping benchmark_wolverine")
    endif()
    foreach(engine IN LISTS native_engines)
        add_executable(benchmark_${engine} run/benchmark_${engine}.cpp)
        target_link_libraries(benchmark_${engine} PRIVATE paper_support paper_native_initial)
        target_include_directories(benchmark_${engine} PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/headers/${engine}")
    endforeach()
else()
    message(STATUS "hnswlib sources absent; building VIGOR only. Fetch refs and reconfigure for baselines.")
endif()
